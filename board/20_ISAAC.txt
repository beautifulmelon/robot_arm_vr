--- 부탁: 식별자를 1순위로, 관절 수는 폴백으로 ---

  A-10 으로 이미 넣으신 **팔 식별자(robot)** 를 권위로 쓰면 정확해집니다.

      def check_robot_match(state, urdf_name, dof):
          if state is None or not state.q:
              return None
          mine = Path(urdf_name).name
          # ★ 식별자가 있으면 그것만 본다. 관절 수는 손이 붙으면 달라진다.
          if state.robot:
              return None if Path(state.robot).name == mine else \
                     f"받는 쪽은 '{Path(state.robot).name}' 인데 이 설정은 '{mine}' 입니다"
          # 식별자가 없을 때만 관절 수로 판단 (구버전 호환)
          if len(state.q) != dof:
              return f"받는 쪽은 관절 {len(state.q)}개인데 이 설정은 {dof}-DOF ({mine}) 입니다"
          return None

        def wrist_axis(ik, cfg):
            """마지막 관절을 조금 돌려 EE 로컬 회전축을 실측한다."""
            q0 = np.asarray(cfg.home, dtype=float); ik.set_q(q0); T0 = ik.fk()
            q1 = q0.copy(); q1[-1] += np.radians(5.0); ik.set_q(q1); T1 = ik.fk()
            Rr = T0[:3,:3].T @ T1[:3,:3]
            ang = np.arccos(np.clip((np.trace(Rr)-1)/2, -1, 1))
            ax = np.array([Rr[2,1]-Rr[1,2], Rr[0,2]-Rr[2,0], Rr[1,0]-Rr[0,1]]) / (2*np.sin(ang))
            ik.set_q(q0); return ax / np.linalg.norm(ax)

        ROLL_AXIS  = wrist_axis(ik, cfg)                              # 기동 시 1회
        ROLL_LIMIT = min(abs(cfg.lower[-1]), cfg.upper[-1])           # = 178°

        434행   wrist_roll = float(np.clip(wrist_roll + right.stick[0] * 0.03,
                                           -ROLL_LIMIT, ROLL_LIMIT))
                # 지금은 ±1.57 rad(±90°) 가 박혀 있습니다 → 사용자 요청대로 ±178°

        458행   roll = rotvec_to_rotation(ROLL_AXIS * wrist_roll)
                # 지금은 np.array([wrist_roll, 0, 0]) — 엉뚱한 X축
